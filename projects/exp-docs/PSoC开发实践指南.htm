单核烧录约定

为了方便单串口调试设备信息，避免手中没有额外 UART 外设串口芯片和杜邦线的开发者难以调试，本项目采用 single burn 单核整合包思想：弃用双核分别烧录方式，将传感器读取、LVGL 显示、KEY/LED、报警音等业务统一整合到 M55 工程中，并使用单 UART 串口输出调试日志。

本文四个工程实际提交和烧录时均只使用一个 RT-Thread Studio 工程，编译输出统一为 Debug/rtthread.elf。

前言说明

本工程实现 AHT20 温湿度采样、LVGL 图形显示、触摸暂停、KEY 中断、三色 LED 状态指示。工程基于 BSP 示例工程创建基础原型，并在应用层完成传感器读取、界面刷新和按键/LED 控制。

功能概览

本工程用于环境温湿度监测：M55 通过 I2C2 读取板载 AHT20，将温度、湿度、舒适度等级显示到 LVGL 触摸屏；实体 KEY 和触摸按钮都可以暂停/恢复采样；三色 LED 根据舒适度给出状态提示。

AHT20 是数字温湿度传感器，培训手册“驱动温湿度传感器”部分说明其固定 7 位 I2C 地址为 0x38，软复位命令为 0xBA，触发测量命令为 0xAC。传感器测量需要时间，主机发出 0xAC 后应延时 75ms 以上再读取状态和数据。

AHT20 返回 6 字节数据，其中 data[0] 是状态字节：bit7 为 BUSY，bit3 表示校准状态。有效数据由 20 位湿度原始值和 20 位温度原始值组成，常用换算公式如下。

humidity_raw = (data[1] << 12) | (data[2] << 4) | ((data[3] & 0xF0) >> 4);
temp_raw     = ((data[3] & 0x0F) << 16) | (data[4] << 8) | data[5];

RH = humidity_raw * 100 / 1048576;
T  = temp_raw * 200 / 1048576 - 50;

本工程在 RT-Thread Settings 中打开 I2C 与 SOFT_I2C2，rtconfig.h 中应能看到 BSP_SOFT_I2C2_SCL_PIN = 75、BSP_SOFT_I2C2_SDA_PIN = 74，以及 PKG_AHT10_I2C_BUS_NAME = "i2c2"。RT-Thread 编程指南说明 I2C 总线以设备名注册，上层通过名称查找总线并进行传输，因此应用层直接使用 aht10_init("i2c2") 即可绑定该总线。

提示：本工程不使用硬件 PWM 控制 LED。LED 闪烁由 thermometer_led_update 中的 rt_tick_get()/rt_tick_from_millisecond() 实现。

3. 替换 applications/exam_shared.h

进入工程目录下的 applications 文件夹。如果没有 exam_shared.h，则新建该文件；如果已经存在，则打开后全篇替换为下面代码。

#ifndef EXAM_SHARED_H__
#define EXAM_SHARED_H__

#include <rtthread.h>

#define THERMO_TEMP_WARM_C        28.0f
#define THERMO_HUMI_WARM_PERCENT  70.0f
#define THERMO_TEMP_ALARM_C       35.0f
#define THERMO_HUMI_ALARM_PERCENT 80.0f
#define THERMO_TEMP_COOL_C        16.0f
#define THERMO_HUMI_DRY_PERCENT   30.0f

typedef enum
{
    THERMO_COMFORT_GOOD = 0,      /* 温湿度舒适：绿灯慢闪 */
    THERMO_COMFORT_WARM,          /* 偏热或偏潮：红灯慢闪 */
    THERMO_COMFORT_ALARM,         /* 高温或高湿报警：红灯常亮 */
    THERMO_COMFORT_DRY_COOL       /* 偏干或偏冷：蓝灯慢闪 */
} thermo_comfort_level_t;

static inline thermo_comfort_level_t thermo_get_comfort_level(float temp, float humi)
{
    if (temp >= THERMO_TEMP_ALARM_C || humi >= THERMO_HUMI_ALARM_PERCENT)
    {
        return THERMO_COMFORT_ALARM;
    }
    if (temp >= THERMO_TEMP_WARM_C || humi >= THERMO_HUMI_WARM_PERCENT)
    {
        return THERMO_COMFORT_WARM;
    }
    if (temp < THERMO_TEMP_COOL_C || humi < THERMO_HUMI_DRY_PERCENT)
    {
        return THERMO_COMFORT_DRY_COOL;
    }

    return THERMO_COMFORT_GOOD;
}
typedef struct
{
    rt_uint32_t magic;       /* 固定标记：说明本缓存已经初始化 */
    rt_uint32_t sequence;    /* 每采样一次加 1，UI 用它显示刷新次数 */
    float temperature;       /* AHT20 温度，单位摄氏度 */
    float humidity;          /* AHT20 湿度，单位 %RH */
    rt_uint8_t paused;       /* 按键或触摸切换：1 表示暂停采样/刷新 */
    rt_uint8_t alarm;        /* 1 表示超过阈值，可用于 LED 或 UI 告警 */
    rt_uint8_t sensor_ok;    /* 1 表示 AHT20 初始化成功 */
    rt_uint8_t reserved;
} thermometer_shared_t;

#define THERMOMETER_SHARED_MAGIC 0x54484D31u

extern volatile thermometer_shared_t g_thermometer_shared;

#endif

4. 替换 applications/main.c

继续在 applications 文件夹中打开 main.c，将文件内容全篇替换为下面代码。该文件负责 AHT20 初始化、周期采样、实体 KEY 中断、三色 LED 状态刷新。

#include <rtthread.h>
#include <rtdevice.h>
#include <board.h>
#include <lv_rt_thread_conf.h>
#include "lv_port_disp.h"
#include "aht10.h"
#include "exam_shared.h"

#define LED_PIN_B                 GET_PIN(16, 5)
#define LED_PIN_G                 GET_PIN(16, 6)
#define LED_PIN_R                 GET_PIN(16, 7)
#define LED_ON_LEVEL              PIN_HIGH
#define LED_OFF_LEVEL             PIN_LOW
#define BUTTON_PIN                GET_PIN(8, 3)
#define SAMPLE_PERIOD_MS          1000
#define LED_REFRESH_MS            10

volatile thermometer_shared_t g_thermometer_shared =
{
    THERMOMETER_SHARED_MAGIC,
    0,
    0.0f,
    0.0f,
    0,
    0,
    0,
    0
};

static aht10_device_t g_aht10_dev = RT_NULL;

void ui_thermometer_init(void);

static void thermometer_shared_reset(void)
{
    /* 单烧录工程不再依赖跨核共享区，启动时明确初始化 UI 数据缓存。 */
    g_thermometer_shared.magic = THERMOMETER_SHARED_MAGIC;
    g_thermometer_shared.sequence = 0;
    g_thermometer_shared.temperature = 0.0f;
    g_thermometer_shared.humidity = 0.0f;
    g_thermometer_shared.paused = 0;
    g_thermometer_shared.alarm = 0;
    g_thermometer_shared.sensor_ok = 0;
    g_thermometer_shared.reserved = 0;
}

void lv_user_gui_init(void)
{
    ui_thermometer_init();
}

static float thermo_absf(float value)
{
    return value < 0.0f ? -value : value;
}

static void thermometer_key_callback(void *args)
{
    (void)args;

    /* 中断里只翻转暂停标志，主循环负责所有较慢的传感器读数工作。 */
    g_thermometer_shared.paused = !g_thermometer_shared.paused;
}

static void thermometer_led_all_off(void)
{
    rt_pin_write(LED_PIN_R, LED_OFF_LEVEL);
    rt_pin_write(LED_PIN_G, LED_OFF_LEVEL);
    rt_pin_write(LED_PIN_B, LED_OFF_LEVEL);
}

static void thermometer_led_update(float temperature, float humidity)
{
    rt_tick_t now;
    rt_uint8_t blink_on;
    thermo_comfort_level_t level;

    /* 板载三色灯按 HIGH 点亮、LOW 熄灭处理。
     * 每 10ms 重写一次三灯状态，压住 M33 残留程序的蓝灯心跳。 */
    now = rt_tick_get();
    blink_on = ((now / rt_tick_from_millisecond(1000)) % 2) ? 1 : 0;
    level = thermo_get_comfort_level(temperature, humidity);

    thermometer_led_all_off();

    switch (level)
    {
    case THERMO_COMFORT_ALARM:
        rt_pin_write(LED_PIN_R, LED_ON_LEVEL);
        break;
    case THERMO_COMFORT_WARM:
        rt_pin_write(LED_PIN_R, blink_on ? LED_ON_LEVEL : LED_OFF_LEVEL);
        break;
    case THERMO_COMFORT_DRY_COOL:
        rt_pin_write(LED_PIN_B, blink_on ? LED_ON_LEVEL : LED_OFF_LEVEL);
        break;
    case THERMO_COMFORT_GOOD:
    default:
        rt_pin_write(LED_PIN_G, blink_on ? LED_ON_LEVEL : LED_OFF_LEVEL);
        break;
    }
}

static void thermometer_sample_once(void)
{
    float temperature;
    float humidity;
    rt_uint8_t alarm;

    if (g_aht10_dev == RT_NULL)
    {
        return;
    }

    humidity = aht10_read_humidity(g_aht10_dev);
    temperature = aht10_read_temperature(g_aht10_dev);
    alarm = (thermo_get_comfort_level(temperature, humidity) == THERMO_COMFORT_ALARM) ? 1 : 0;

    g_thermometer_shared.temperature = temperature;
    g_thermometer_shared.humidity = humidity;
    g_thermometer_shared.alarm = alarm;
    g_thermometer_shared.sensor_ok = 1;
    g_thermometer_shared.sequence++;

    rt_kprintf("[Single Thermo] temp=%d.%d C, humi=%d.%d %%, alarm=%d, seq=%u\r\n",
               (int)temperature, (int)(thermo_absf(temperature) * 10) % 10,
               (int)humidity, (int)(thermo_absf(humidity) * 10) % 10,
               alarm, g_thermometer_shared.sequence);
}

int main(void)
{
    rt_kprintf("Single Thermometer M55: AHT20 + LVGL + key/LED\r\n");
    thermometer_shared_reset();

    rt_pin_mode(LED_PIN_R, PIN_MODE_OUTPUT);
    rt_pin_mode(LED_PIN_G, PIN_MODE_OUTPUT);
    rt_pin_mode(LED_PIN_B, PIN_MODE_OUTPUT);
    thermometer_led_all_off();
    rt_pin_mode(BUTTON_PIN, PIN_MODE_INPUT_PULLUP);
    rt_pin_attach_irq(BUTTON_PIN, PIN_IRQ_MODE_FALLING, thermometer_key_callback, RT_NULL);
    rt_pin_irq_enable(BUTTON_PIN, PIN_IRQ_ENABLE);

    lvgl_thread_init();
    rt_thread_mdelay(2000);

    g_aht10_dev = aht10_init(PKG_AHT10_I2C_BUS_NAME);
    if (g_aht10_dev == RT_NULL)
    {
        g_thermometer_shared.sensor_ok = 0;
        rt_kprintf("[Single Thermo] AHT20 init failed on %s\r\n", PKG_AHT10_I2C_BUS_NAME);
    }
    else
    {
        g_thermometer_shared.sensor_ok = 1;
        rt_kprintf("[Single Thermo] AHT20 ready on %s\r\n", PKG_AHT10_I2C_BUS_NAME);
    }

    while (1)
    {
        static rt_uint32_t sample_elapsed = SAMPLE_PERIOD_MS;

        if (!g_thermometer_shared.paused && sample_elapsed >= SAMPLE_PERIOD_MS)
        {
            thermometer_sample_once();
            sample_elapsed = 0;
        }

        thermometer_led_update(g_thermometer_shared.temperature, g_thermometer_shared.humidity);
        rt_thread_mdelay(LED_REFRESH_MS);
        sample_elapsed += LED_REFRESH_MS;
    }
}

{
    return value < 0 ? -value : value;
}

static void ui_set_label_x10(lv_obj_t *label, float value, const char *unit)
{
    char buf[32];
    int scaled;
    int whole;
    int frac;

    /* 当前工具链的 LVGL printf 未开启浮点格式化，因此先转成十分位整数再显示。 */
    if (value >= 0.0f)
    {
        scaled = (int)(value * 10.0f + 0.5f);
    }
    else
    {
        scaled = (int)(value * 10.0f - 0.5f);
    }

    whole = scaled / 10;
    frac = ui_abs_int(scaled % 10);

    if (scaled < 0 && whole == 0)
    {
        lv_snprintf(buf, sizeof(buf), "-%d.%d %s", ui_abs_int(whole), frac, unit);
    }
    else
    {
        lv_snprintf(buf, sizeof(buf), "%d.%d %s", whole, frac, unit);
    }

    lv_label_set_text(label, buf);
}

static const char *ui_comfort_text(float temp, float humi)
{
    switch (thermo_get_comfort_level(temp, humi))
    {
    case THERMO_COMFORT_ALARM:
        return "ALARM  High temp/humi";
    case THERMO_COMFORT_WARM:
        return "WARM   Ventilate";
    case THERMO_COMFORT_DRY_COOL:
        return "DRY/COOL  Check room";
    case THERMO_COMFORT_GOOD:
    default:
        return "COMFORT  Good";
    }
}

static void ui_set_state_color(rt_uint8_t sensor_ok, rt_uint8_t paused, rt_uint8_t alarm)

static void ui_timer_cb(lv_timer_t *timer)
{
    const volatile thermometer_shared_t *shared = &g_thermometer_shared;
    char buf[32];

    (void)timer;

    /* 这个工程由 M55 本核通过 I2C 直接读取 AHT20，不再等待跨核传感器数据。 */
    ui_set_label_x10(s_temp_value, shared->temperature, "C");
    ui_set_label_x10(s_humi_value, shared->humidity, "%");
    lv_label_set_text(s_comfort_label, ui_comfort_text(shared->temperature, shared->humidity));

    if (!shared->sensor_ok)
    {
        lv_label_set_text(s_state_label, "AHT20 OFFLINE / RETRY");
    }
    else if (shared->paused)
    {
        lv_label_set_text(s_state_label, "PAUSED BY KEY/TOUCH");
    }

硬件依据

AHT20 与前一章使用同一颗板载温湿度传感器，查看 PSoc_Edge_Basic_Schematic.pdf 第 12 页 Sensor。其 SCL/SDA 连接到 RPI_SCL/SPI1_CLK 与 RPI_SDA/SPI1_MOSI 网络；工程配置中对应软件 I2C2，SCL pin number 为 75，SDA pin number 为 74。

图：PSoc_Edge_Basic_Schematic.pdf 第 12 页，AHT20 温湿度传感器与 LSM6DS3 原理图。

培训手册中 AHT20 小节给出的关键参数为：固定 7 位 I2C 地址 0x38，软复位 0xBA，触发测量 0xAC；触发测量后延时 75ms 以上再读取。XiaoZhi 扩展代码按这个流程手写 I2C 命令，避免依赖浮点 printf 或额外示例线程。

本工程没有直接使用 %f 显示浮点数，而是把温度扩大 10 倍保存为 temp_x10，再用整数拆出个位和一位小数。AHT20 原始数据处理公式如下，1048576 即 2 的 20 次方。

humidity_raw = (data[1] << 12) | (data[2] << 4) | ((data[3] & 0xF0) >> 4);
temp_raw     = ((data[3] & 0x0F) << 16) | (data[4] << 8) | data[5];

temp_calc_x10 = (temp_raw * 2000 + 524288) / 1048576 - 500;
humidity_calc = (humidity_raw * 100 + 524288) / 1048576;

// temp_calc_x10 = 253 表示 25.3C；humidity_calc = 55 表示 55%RH。

XiaoZhi 原工程还需要 Audio 和 WiFi。Audio 硬件见 Basic schematic 第 1 页：ES8388 通过 I2S_TX_MCK/SCK/FSYNC/SD 与主控传输音频，通过 I2C_SCL/I2C_SDA 配置寄存器，输出到 SPK_IN 后接功放和扬声器接口。

图：PSoc_Edge_Basic_Schematic.pdf 第 1 页，ES8388 音频编解码器、SPK_IN 与扬声器功放链路。

#define XZ_AHT20_CMD_SOFT_RESET     0xBA

static rt_err_t xz_aht20_send3(struct rt_i2c_bus_device *bus,
                               rt_uint8_t cmd,
                               rt_uint8_t arg0,
                               rt_uint8_t arg1)
{
    rt_uint8_t packet[3];

    packet[0] = cmd;
    packet[1] = arg0;
    packet[2] = arg1;

    return (rt_i2c_master_send(bus, XZ_AHT20_ADDR, 0, packet, sizeof(packet)) == (rt_ssize_t)sizeof(packet)) ?
           RT_EOK : -RT_ERROR;
}

static rt_err_t xz_aht20_send1(struct rt_i2c_bus_device *bus, rt_uint8_t cmd)
{
    return (rt_i2c_master_send(bus, XZ_AHT20_ADDR, 0, &cmd, 1) == 1) ? RT_EOK : -RT_ERROR;
}

static rt_err_t xz_aht20_read(struct rt_i2c_bus_device *bus, rt_uint8_t *data, rt_uint32_t len)
{
    return (rt_i2c_master_recv(bus, XZ_AHT20_ADDR, 0, data, len) == (rt_ssize_t)len) ?
           RT_EOK : -RT_ERROR;
}

static rt_bool_t xz_aht20_decode(const rt_uint8_t data[6], int *temp_x10, int *humi_pct)
{
    rt_uint32_t humidity_raw;
    rt_uint32_t temp_raw;
    rt_int32_t temp_calc_x10;
    rt_uint32_t humidity_calc;

    if ((data[0] & XZ_AHT20_STATUS_BUSY) != 0)
    {
        return RT_FALSE;
    }

    humidity_raw = ((rt_uint32_t)data[1] << 12) |
                   ((rt_uint32_t)data[2] << 4) |
                   (((rt_uint32_t)data[3] & 0xF0) >> 4);
    temp_raw = (((rt_uint32_t)data[3] & 0x0F) << 16) |
               ((rt_uint32_t)data[4] << 8) |
               (rt_uint32_t)data[5];

    temp_calc_x10 = (rt_int32_t)((temp_raw * 2000U + 524288U) / 1048576U) - 500;
    humidity_calc = (humidity_raw * 100U + 524288U) / 1048576U;
    if (humidity_calc > 100U)
    {
        humidity_calc = 100U;
    }

    *temp_x10 = (int)temp_calc_x10;
    *humi_pct = (int)humidity_calc;
    return RT_TRUE;
}

static rt_bool_t xz_aht20_read_once(struct rt_i2c_bus_device *bus, int *temp_x10, int *humi_pct)
{
    rt_uint8_t status = 0;
    rt_uint8_t data[6] = {0};

    if (xz_aht20_send3(bus, XZ_AHT20_CMD_INIT, 0x08, 0x00) != RT_EOK)
    {
        xz_aht20_send1(bus, XZ_AHT20_CMD_SOFT_RESET);
        rt_thread_mdelay(20);
        if (xz_aht20_send3(bus, XZ_AHT20_CMD_INIT, 0x08, 0x00) != RT_EOK)
        {
            return RT_FALSE;
        }
    }
    rt_thread_mdelay(10);

    if (xz_aht20_read(bus, &status, 1) != RT_EOK)
    {
        return RT_FALSE;
    }

    if ((status & XZ_AHT20_STATUS_CALIBRATED) == 0)
    {
        return RT_FALSE;
    }

    if (xz_aht20_send3(bus, XZ_AHT20_CMD_TRIGGER, 0x33, 0x00) != RT_EOK)
    {
        return RT_FALSE;
    }
    rt_thread_mdelay(80);

    if (xz_aht20_read(bus, data, sizeof(data)) != RT_EOK)
    {
        return RT_FALSE;
    }

    return xz_aht20_decode(data, temp_x10, humi_pct);
}

static void xz_temp_humi_thread(void *parameter)
{
    struct rt_i2c_bus_device *bus;
    char text[64];

    (void)parameter;

    /* 等待 XiaoZhi UI 和软件 I2C 驱动完成正常启动。 */
    rt_thread_mdelay(2500);

    bus = rt_i2c_bus_device_find(XZ_AHT20_I2C_BUS_NAME);
    if (bus == RT_NULL)
    {
        xiaozhi_ui_set_temp_humi("AHT20: i2c2 missing");
        rt_kprintf("[XiaoZhi Env] AHT20 I2C bus %s not found\r\n", XZ_AHT20_I2C_BUS_NAME);
        return;
    }

    rt_kprintf("[XiaoZhi Env] AHT20 ready on %s\r\n", XZ_AHT20_I2C_BUS_NAME);

    while (1)
    {
        int temp_x10 = 0;
        int humi_pct = 0;
        int temp_abs_x10;

        if (xz_aht20_read_once(bus, &temp_x10, &humi_pct))
        {
            temp_abs_x10 = temp_x10 < 0 ? -temp_x10 : temp_x10;
            rt_snprintf(text, sizeof(text), "T %s%d.%dC  H %d%%",
                        temp_x10 < 0 ? "-" : "",
                        temp_abs_x10 / 10,
                        temp_abs_x10 % 10,
                        humi_pct);
            xiaozhi_ui_set_temp_humi(text);
            rt_kprintf("[XiaoZhi Env] %s\r\n", text);
        }
        else
        {
            xiaozhi_ui_set_temp_humi("AHT20: retry");
            rt_kprintf("[XiaoZhi Env] AHT20 read failed, retry\r\n");
        }

        rt_thread_mdelay(XZ_ENV_SAMPLE_PERIOD_MS);
    }
}

int xz_temp_humi_init(void)
{
    rt_thread_t tid;

    tid = rt_thread_create("xz_env", xz_temp_humi_thread, RT_NULL, 2048, 24, 20);
    if (tid == RT_NULL)
    {
        rt_kprintf("[XiaoZhi Env] create thread failed\r\n");
        return -RT_ERROR;
    }

    rt_thread_startup(tid);
    return RT_EOK;
}

/**
 * @brief 向 UI 消息队列发送消息
 * @param cmd 命令类型
 * @param data 数据字符串，可以为 NULL
 * @param default_data data 为 NULL 时使用的默认值
 */
static void ui_send_message(ui_cmd_t cmd, const char *data, const char *default_data)
{
    ui_msg_t msg;

    msg.cmd = cmd;
    if (data != RT_NULL)
    {
        rt_strncpy(msg.data, data, sizeof(msg.data) - 1);
    }
    else if (default_data != RT_NULL)
    {
        rt_strncpy(msg.data, default_data, sizeof(msg.data) - 1);
    }
    else
    {
        msg.data[0] = '\0';
    }
    msg.data[sizeof(msg.data) - 1] = '\0';

    rt_mq_send(&s_ui_mq, &msg, sizeof(msg));
}

/**
 * @brief 处理 UI 消息
 * @param msg 待处理的消息
 */
static void ui_process_message(const ui_msg_t *msg)
{
    switch (msg->cmd)
    {
    case UI_CMD_SET_STATUS:
        if (s_label_status)
        {
            lv_label_set_text(s_label_status, msg->data);
        }
        break;

    case UI_CMD_SET_OUTPUT:
        if (s_label_output)
        {
            lv_label_set_text(s_label_output, msg->data);
        }
        break;

    case UI_CMD_SET_ADC:
        if (s_label_adc)
        {
            lv_label_set_text(s_label_adc, msg->data);
        }
        break;

    case UI_CMD_SET_EMOJI:
        ui_show_emoji(ui_find_emoji_index(msg->data));
        break;

    case UI_CMD_CLEAR_INFO:
        if (s_label_info) lv_label_set_text(s_label_info, " ");
        if (s_label_output) lv_label_set_text(s_label_output, " ");
        break;

    case UI_CMD_SHOW_AP_INFO:
        if (s_label_status) lv_label_set_text(s_label_status, "Connecting...");
        if (s_label_info) lv_label_set_text(s_label_info, "Connect phone/PC to hotspot");
        if (s_label_output) lv_label_set_text(s_label_output, msg->data[0] ? msg->data : AP_INFO_DEFAULT_TEXT);
        break;

    case UI_CMD_SHOW_CONNECTING:
        if (s_label_status) lv_label_set_text(s_label_status, "Connecting...");
        if (s_label_info) lv_label_set_text(s_label_info, "Connecting saved WiFi...");
        if (s_label_output) lv_label_set_text(s_label_output, " ");
        break;

    case UI_CMD_UPDATE_BATTERY:
    {
        int level = atoi(msg->data);
        update_battery_display(level);
    }
    break;

    case UI_CMD_UPDATE_CHARGE_STATUS:
        if (strcmp(msg->data, "charging") == 0)
        {
            /* 更新 UI，显示充电状态 */
            if (s_label_status)
            {
                const char *current_text = lv_label_get_text(s_label_status);
                char new_text[128];
                rt_snprintf(new_text, sizeof(new_text), "%s [鍏呯數涓璢", current_text);
                lv_label_set_text(s_label_status, new_text);
            }
        }
        else
        {
            /* 移除充电提示 */
            if (s_label_status)
            {
                const char *current_text = lv_label_get_text(s_label_status);
                char *charging_pos = rt_strstr(current_text, " [鍏呯數涓璢");
                if (charging_pos)
                {
                    char new_text[128];
                    int len = charging_pos - current_text;
                    rt_memcpy(new_text, current_text, len);
                    new_text[len] = '\0';
                    lv_label_set_text(s_label_status, new_text);
                }
            }
        }
        break;

    /* UI_CMD_UPDATE_BLE_STATUS：当前未使用蓝牙 */
    /* case UI_CMD_UPDATE_BLE_STATUS: */
    /*     if (s_img_ble) { */
    /*         if (strcmp(msg->data, "open") == 0) { */
    /*             lv_img_set_src(s_img_ble, &ble); */
    /*         } else { */
    /*             lv_img_set_src(s_img_ble, &ble_close); */
    /*         } */
    /*     } */
    /*     break; */

    default:
        LOG_W("Unknown UI command: %d", msg->cmd);
        break;
    }
}

/**
 * @brief UI 线程入口
 * @param args 线程参数，当前未使用
 */
static void ui_thread_entry(void *args)
{
    ui_msg_t msg;
    rt_uint32_t period_ms;
    lv_font_t *font_36;
    lv_font_t *font_28;
    lv_font_t *font_22;

    (void)args;

    /* 初始化 LVGL */
    lv_init();
    lv_tick_set_cb(&rt_tick_get_millisecond);
    lv_port_disp_init();

    extern void lv_port_indev_init(void);
    lv_port_indev_init();

    /* 获取屏幕分辨率，用于缩放 */
    lv_coord_t scr_width = lv_disp_get_hor_res(NULL);
    lv_coord_t scr_height = lv_disp_get_ver_res(NULL);
    LOG_I("Screen resolution: %d x %d", scr_width, scr_height);

    /* 计算缩放系数 */
    g_scale = get_scale_factor();
    LOG_I("Scale factor: %.2f", g_scale);

    /* 根据 800x512 分辨率初始化缩放后的字体样式 */
    lv_style_init(&s_style_30);
    font_36 = lv_tiny_ttf_create_data(xiaozhi_font, xiaozhi_font_size, (int)(36 * g_scale));
    lv_style_set_text_font(&s_style_30, font_36);
    lv_style_set_text_align(&s_style_30, LV_TEXT_ALIGN_CENTER);
    lv_style_set_text_color(&s_style_30, lv_color_hex(0xffffff));

    lv_style_init(&s_style_24);
    font_28 = lv_tiny_ttf_create_data(xiaozhi_font, xiaozhi_font_size, (int)(28 * g_scale));
    lv_style_set_text_font(&s_style_24, font_28);
    lv_style_set_text_align(&s_style_24, LV_TEXT_ALIGN_CENTER);
    lv_style_set_text_color(&s_style_24, lv_color_hex(0xffffff));

    lv_style_init(&s_style_20);
    font_22 = lv_tiny_ttf_create_data(xiaozhi_font, xiaozhi_font_size, (int)(22 * g_scale));
    lv_style_set_text_font(&s_style_20, font_22);
    lv_style_set_text_align(&s_style_20, LV_TEXT_ALIGN_CENTER);
    lv_style_set_text_color(&s_style_20, lv_color_hex(0xffffff));

    /* 初始化 UI 对象 */
    if (ui_objects_init() != RT_EOK)
    {
        LOG_E("UI objects init failed");
        return;
    }

    /* 设置初始显示内容 */
    if (s_label_status) lv_label_set_text(s_label_status, "鍒濆鍖栦腑...");
    if (s_label_info) lv_label_set_text(s_label_info, " ");
    if (s_label_adc) lv_label_set_text(s_label_adc, " ");
    if (s_label_output) lv_label_set_text(s_label_output, " ");
    ui_show_emoji(0);
    lv_task_handler();

    /* 通知其它线程：UI 初始化完成 */
    rt_sem_release(&s_ui_init_sem);
    LOG_I("UI initialized for 800x512 resolution");

    /* UI 主循环 */
    while (1)
    {
        if (rt_mq_recv(&s_ui_mq, &msg, sizeof(msg), RT_WAITING_NO) > 0)
        {
            ui_process_message(&msg);
        }

        period_ms = lv_task_handler();
        if(period_ms == LV_NO_TIMER_READY || period_ms > 100)
            period_ms = 20;

        rt_thread_mdelay(period_ms);
    }
}

    {
        LOG_E("Create UI thread failed");
    }

核心判定逻辑

摔倒不使用 AI 模型，按数值变化规律做条件匹配：先检测快速运动/冲击，再结合姿态变化与静止稳定阶段确认 FALL；同时排除“手动快速扶正到开发板标准直立姿态”的场景。

串口和界面中的 ACC 表示 accelerometer，即三轴加速度 X/Y/Z，单位为 mg，1000mg 约等于 1g 重力加速度。ACC 主要反映设备姿态、冲击和短暂失重，例如静止时三轴合成模长应接近 1000mg。GYR 表示 gyroscope，即三轴角速度 X/Y/Z，底层变量 angular_rate_mdps 的单位为 mdps，界面显示时换算为 dps；GYR 主要反映设备旋转速度，用于区分普通静止、手动晃动和快速翻倒。

采样线程每 100ms 读取一次 LSM6DS3TR-C。为了降低 MCU 计算量，代码没有对三轴加速度做平方根，而是使用 acc_mag2 = ax*ax + ay*ay + az*az 与阈值平方比较；角速度使用 gyro_sum = |gx| + |gy| + |gz| 做合成强度判断。

状态机并不是检测到一次峰值就立刻报警。程序先记录 drop_dip、impact_rotate、impact_high + fast_rotate 或 posture_change 这类候选事件；候选事件出现后打开 12 个采样周期的确认窗口。如果窗口内设备连续 4 次满足静止条件，才确认 FALL，并保持约 50 个采样周期，约 5 秒。这样可以减少普通手摇造成的误报。

上电后的前 10 个采样周期作为预热阶段，状态直接按 NORMAL 处理，避免传感器刚启动时的全 0 或异常初值触发报警。对于用户手动把板子快速摆回标准直立姿态的情况，fallcheck_is_standard_upright 会额外排除：X 轴大于 850mg，Y/Z 轴绝对值小于 250mg，且角速度总量低于静止阈值时，不确认 FALL。

static rt_bool_t fallcheck_is_standard_upright(float ax, float ay, float az, float gyro_sum)
{
    /*
     * 手动把开发板快速扶回标准直立姿态时：
     * X 轴接近 +1g，Y/Z 轴接近 0，并且陀螺仪已经安静。
     * 这种情况按“安全落点”处理，不作为摔倒确认。
     */
    return (ax > FALLCHECK_UPRIGHT_X_MIN_MG &&
            fallcheck_absf(ay) < FALLCHECK_UPRIGHT_SIDE_MAX_MG &&
            fallcheck_absf(az) < FALLCHECK_UPRIGHT_SIDE_MAX_MG &&
            gyro_sum < FALLCHECK_IDLE_GYRO_SUM_MDPS) ? RT_TRUE : RT_FALSE;
}

报警声通过 wavplayer 播放 ROMFS 中的 beep.wav。工程把原始 WAV 文件转换成 C 数组，放入 applications/webnet/romfs.c，编译时随固件写入 Flash；运行时 fallcheck_romfs_prepare 挂载 ROMFS，打开 /webnet/beep.wav 并读取前 4 字节检查是否为 RIFF 格式，确认资源有效后再允许播放。

报警线程每 100ms 轮询一次共享状态。只有 state 为 FALL、alarm 置位且 muted 为 0 时，才调用 wavplayer_play("/webnet/beep.wav")；同时使用 2500ms 冷却时间，避免同一段报警被过度重复触发。KEY 或触摸静音按钮会切换 muted 标志，静音后仍可在界面看到 FALL 状态，但不会继续播放报警音。

static void fallcheck_led_update(void)
{
    rt_uint8_t state = g_fallcheck_shared.state;

    /* 每次刷新都重写三色灯，避免其它残留逻辑留下错误灯色。 */
    fallcheck_led_all_off();

    if (state == FALLCHECK_STATE_FALL)
    {
        rt_pin_write(LED_PIN_R, LED_ON_LEVEL);
    }
    else if (state == FALLCHECK_STATE_NORMAL)
    {
        rt_pin_write(LED_PIN_B, LED_ON_LEVEL);
    }
    else
    {
        rt_pin_write(LED_PIN_G, LED_ON_LEVEL);
    }
}

static void fallcheck_alarm_task(void *parameter)
{
    rt_tick_t last_play_tick = 0;
    rt_uint8_t was_fall = 0;
    rt_uint8_t romfs_ok;

    (void)parameter;

    romfs_ok = (fallcheck_romfs_prepare() == RT_EOK) ? 1 : 0;

    while (1)
    {
        rt_uint8_t is_fall = (g_fallcheck_shared.state == FALLCHECK_STATE_FALL &&
                              g_fallcheck_shared.alarm &&
                              !g_fallcheck_shared.muted) ? 1 : 0;
        rt_tick_t now = rt_tick_get();
        rt_tick_t cooldown = rt_tick_from_millisecond(FALLCHECK_BEEP_COOLDOWN_MS);

        if (romfs_ok && is_fall && (!was_fall || (now - last_play_tick) >= cooldown))
        {
            last_play_tick = now;
            if (wavplayer_play(FALLCHECK_BEEP_PATH) != 0)
            {
                rt_kprintf("[Single FallCheck] wavplayer failed: %s\r\n", FALLCHECK_BEEP_PATH);
            }
        }

        was_fall = is_fall;
        rt_thread_mdelay(FALLCHECK_ALARM_POLL_MS);
    }
}

四、温湿度与摔倒检测融合实践：ThermoFall

前言说明

ThermoFall 是四个工程中最复杂的融合工程，由 Single_Thermo_AllIn 和 Single_Fall_AllIn 合并而来。它在一个 M55 工程里同时运行 AHT20 温湿度任务和 LSM6DS3 摔倒检测任务，通过 LVGL 触控按钮切换 Thermo/Fall 页面。

提示：本章只给出融合设计和核心代码。完整代码后续由维护者上传 git 仓库，本文档中仅预留链接位置。

功能概览

本工程同时覆盖环境监测和姿态异常检测。Thermo 页面显示 AHT20 温湿度与舒适度；FallCheck 页面显示 LSM6DS3 的 acc/gyr 与 IDLE/MOVING/FALL 状态。触摸按钮只负责页面切换，实体 KEY 根据当前页面分别执行温湿度暂停或 Fall 报警静音。

3.1 ISR、线程与共享变量

1.            ISR 中只读取必要状态、清中断并释放信号量或发送轻量事件；不要延时、申请大块内存、访问文件系统或打印长日志。

2.            volatile 只保证编译器每次访问变量，不提供原子性和互斥；多字节共享状态仍需要临界区、互斥量或消息队列。

3.            线程优先级按实时性设置：采集和告警高于界面，界面高于后台日志；同优先级线程不能依赖偶然调度顺序。

4.            每个线程都检查创建返回值，并通过 list_thread 观察状态、剩余栈和运行情况。

4.2 最小代码骨架

#include <rtthread.h>
#include <rtdevice.h>

#define LED_PIN GET_PIN(16, 7)
#define KEY_PIN GET_PIN(0, 4)       /* 现场按原理图替换 */

static struct rt_semaphore key_sem;

static void key_isr(void *args)
{
    /* ISR 只通知线程；不延时、不打印、不做传感器访问。 */
    rt_sem_release(&key_sem);
}

static void key_thread(void *args)
{
    rt_tick_t press_tick;
    while (1)
    {
        rt_sem_take(&key_sem, RT_WAITING_FOREVER);
        rt_thread_mdelay(20);       /* 线程中消抖 */
        if (rt_pin_read(KEY_PIN) != PIN_LOW) continue;
        press_tick = rt_tick_get();
        while (rt_pin_read(KEY_PIN) == PIN_LOW) rt_thread_mdelay(10);
        if (rt_tick_get() - press_tick >= rt_tick_from_millisecond(800))
            rt_kprintf("KEY LONG\n");
        else
            rt_kprintf("KEY SHORT\n");
    }
}

函数解释： rt_pin_mode 配置方向和上下拉；rt_pin_attach_irq 绑定回调；rt_pin_irq_enable 才真正打开中断；rt_sem_release 可把事件交给线程；rt_tick_get 适合计算持续时间。初始化顺序为 GPIO 模式、信号量、IRQ 回调、使能 IRQ、创建线程。

验收： 连按 20 次只触发 20 次；长按不被误判成多次短按；主循环繁忙时仍能响应；串口日志不出现在 ISR 内。

常见故障： LED 极性相反、按键浮空、边沿选错、中断使能遗漏、机械抖动、按键引脚与下载/串口复用。

5. 定时器与周期任务

5.1 软件定时器

static rt_timer_t sample_timer;
static struct rt_semaphore sample_sem;

static void timer_cb(void *parameter)
{
    rt_sem_release(&sample_sem);    /* 回调只发事件 */
}

static void timer_init(void)
{
    rt_sem_init(&sample_sem, "smp", 0, RT_IPC_FLAG_FIFO);
    sample_timer = rt_timer_create("sample", timer_cb, RT_NULL,
        rt_tick_from_millisecond(1000), RT_TIMER_FLAG_PERIODIC | RT_TIMER_FLAG_SOFT_TIMER);
    RT_ASSERT(sample_timer != RT_NULL);
    rt_timer_start(sample_timer);
}

修改周期时先准备新 tick，再调用 rt_timer_control(timer, RT_TIMER_CTRL_SET_TIME, &ticks)。单次超时使用 RT_TIMER_FLAG_ONE_SHOT。软件定时器用于秒级采样、超时和状态维护；硬件定时器用于更高精度、输入捕获或需要固定抖动上限的任务；PWM 波形必须交给 TCPWM，不用线程翻转 GPIO 伪造。

验收： 运行 60 秒记录触发次数；回调执行时间远小于周期；动态改周期后立即生效；停止定时器后不再产生事件。

6. PWM：呼吸灯与蜂鸣器

6.1 配置与 API

在 Device Configurator 中选择空闲 TCPWM 通道、输出引脚、时钟分频和极性，生成代码后在 RT-Thread Settings 打开 PWM 设备。先用 list_device 确认设备名。

struct rt_device_pwm *pwm;
rt_uint32_t period_ns = 1000000;   /* 1 kHz */
rt_uint32_t duty_pct = 35;

pwm = (struct rt_device_pwm *)rt_device_find("pwm0");
RT_ASSERT(pwm != RT_NULL);
rt_pwm_set(pwm, 1, period_ns, period_ns * duty_pct / 100);
rt_pwm_enable(pwm, 1);

rt_pwm_set(device, channel, period, pulse) 的周期和高电平时间通常使用纳秒，必须满足 0 <= pulse <= period。频率公式为 f = 1e9 / period_ns，占空比为 pulse / period * 100%。

呼吸灯每 10–20 ms 修改一次占空比，到达 0% 或 100% 后反向；蜂鸣器变调通过修改周期实现，不要把音量和频率混为一项。若 PWM 无输出，依次检查设备名、通道号、PinMux、时钟、极性和示波器/逻辑分析仪测量结果。

7. ADC：采样、滤波和阈值回差

7.1 最小读取与换算

#define ADC_CTRL GET_PIN(8, 4)

rt_pin_mode(ADC_CTRL, PIN_MODE_OUTPUT);
rt_pin_write(ADC_CTRL, PIN_HIGH);
rt_adc_device_t adc = (rt_adc_device_t)rt_device_find("adc1");
RT_ASSERT(adc != RT_NULL);
rt_adc_enable(adc, 0);
rt_uint32_t raw = rt_adc_read(adc, 0);
rt_uint32_t mv = (raw * 4200U + 900U) / 1800U;
rt_kprintf("battery raw=%u voltage=%u.%03uV\n", raw, mv / 1000U, mv % 1000U);

以上是官方 Edgi_Talk_M33_ADC_EX 的板载电池采样链路：P8_4 必须拉高，换算式包含板载模拟前端的比例/标定。它不能直接用于外接电位器。外接模拟量必须根据题目给出的参考电压、12 位满量程和分压比换算；高阻信号、参考电压误差、悬空输入和采样时间都会造成偏差。

7.2 八点滑动平均

#define N 8
static rt_uint16_t buf[N];
static rt_uint32_t sum;
static rt_uint8_t pos;

static rt_uint16_t filter_push(rt_uint16_t value)
{
    sum -= buf[pos];
    buf[pos] = value;
    sum += value;
    pos = (pos + 1) % N;
    return (rt_uint16_t)(sum / N);
}

阈值报警要使用回差和连续计数，例如高于 3000 连续 5 次进入报警，低于 2800 连续 5 次解除。这样能避免传感器在边界附近抖动。验收时同时打印原始值和滤波值，快速旋钮应有响应，静止时滤波值更稳定。

8. UART：接收回调、环形缓冲与命令解析

8.1 接收结构

static rt_device_t serial;
static struct rt_semaphore rx_sem;

static rt_err_t uart_rx_ind(rt_device_t dev, rt_size_t size)
{
    rt_sem_release(&rx_sem);
    return RT_EOK;
}

static void uart_thread(void *parameter)
{
    char ch, line[64];
    rt_size_t used = 0;
    while (1)
    {
        rt_sem_take(&rx_sem, RT_WAITING_FOREVER);
        while (rt_device_read(serial, 0, &ch, 1) == 1)
        {
            if (ch == '\r' || ch == '\n')
            {
                if (used) { line[used] = '\0'; command_execute(line); used = 0; }
            }
            else if (used < sizeof(line) - 1) line[used++] = ch;
            else used = 0;        /* 超长行安全丢弃 */
        }
    }
}

初始化时依次查找设备、配置波特率/数据位/停止位/校验、以读写和中断接收方式打开、设置 RT_DEVICE_CTRL_SET_RX_INDICATE 回调。命令解析至少支持 LED ON/OFF、PWM 0..100、PERIOD 100..5000、STATUS 和 HELP，所有数字参数都做范围检查。

验收： Windows 的 CRLF 和单独 LF 都能结束命令；快速粘贴多条命令不会死机；空行、未知命令、非法数字和超长输入都有明确反馈。乱码优先检查波特率、数据位、校验、停止位、时钟和串口是否被 MSH 占用。

9. I2C 与传感器掉线恢复

9.1 寄存器读取

static rt_err_t i2c_read_reg(struct rt_i2c_bus_device *bus,
                             rt_uint8_t addr, rt_uint8_t reg,
                             rt_uint8_t *data, rt_size_t len)
{
    struct rt_i2c_msg msg[2];
    msg[0].addr = addr; msg[0].flags = RT_I2C_WR; msg[0].buf = &reg; msg[0].len = 1;
    msg[1].addr = addr; msg[1].flags = RT_I2C_RD; msg[1].buf = data; msg[1].len = len;
    return rt_i2c_transfer(bus, msg, 2) == 2 ? RT_EOK : -RT_ERROR;
}

I2C 地址在 RT-Thread API 中通常填写 7 位地址，不要把数据手册里的读写位再次左移。扫描总线只能用于调试，正式程序应读取 WHO_AM_I/CHIP_ID 验证器件身份。

9.2 两类硬件配置

1.            Edgi-Talk 实板：板载 AHT20（U18，0x38）和 LSM6DS3TR-C。官方 AHT20 EX 使用软件 i2c1（SCL/SDA 引脚号 75/74）；官方 LSM6DS3 EX 使用硬件 i2c0 和地址 0x6A。

2.            基础 KIT：BMI270 常见 0x68/0x69，寄存器 0x00 应读取到 CHIP_ID 0x24；完整初始化要使用官方驱动加载配置数据。

每次读取都检查返回值和数据范围。连续失败 3 次标记离线，降低重试频率；每 2 秒尝试重新初始化，成功连续 2 次后再恢复在线。I2C NACK 按“供电与共地、SDA/SCL 上拉、PinMux、总线名、7 位地址、时序/重复起始、器件复位”顺序排查。

11. LCD Touch 与麦克风

Edgi-Talk 1.3.1 没有板载 PSoC CapSense 按键或滑条。板上的触摸输入来自 LCD 模组的 ST7102 电容触摸控制器，通过 I2C 访问（BSP 驱动默认地址 0x55）；模式切换基础题优先使用机械用户按键。不要把 ST7102 的 RT-Thread Touch 设备接口与 PSoC CapSense API 混用。

麦克风题优先复用 Edgi_Talk_M33_Audio 的录音设备配置。短时能量可用固定长度 PCM 窗口计算：

static rt_uint32_t audio_rms2(const rt_int16_t *pcm, rt_size_t count)
{
    rt_uint64_t sum = 0;
    for (rt_size_t i = 0; i < count; i++)
        sum += (rt_int32_t)pcm[i] * (rt_int32_t)pcm[i];
    return count ? (rt_uint32_t)(sum / count) : 0;
}

比赛通常不要求浮点开方，只需用均方能量与阈值比较。加入噪声基线、连续窗口计数和饱和检测；音频回调中不要执行整窗计算，交给处理线程。

12. 状态机与数据有效性

推荐状态为 INIT / NORMAL / WARNING / ALARM / SENSOR_OFFLINE。状态转移必须由连续计数和回差控制：进入报警需要连续 3–5 次异常，解除报警需要连续 5 次正常；传感器超时后进入离线，不保留旧值冒充实时数据。

摔倒检测可复用前文“候选冲击或姿态变化 → 确认窗口 → 稳定姿态 → 报警保持”的结构。阈值只是参数，核心是单位换算、时间窗口、连续确认和现场样本验证。

13. 看门狗、内存与实时性优化

1.            看门狗只由确认系统关键任务均健康的监控线程喂；不要在定时中断里无条件喂狗。

2.            所有线程入口不得返回；循环中必须有阻塞点，避免 100% 占用 CPU。

3.            大数组和音频缓存避免放在线程栈；检查 map 文件、heap 使用和线程剩余栈。

5.            测量最坏执行时间和周期抖动，不只观察平均值；日志过多会改变实时性。

1.            0–10 分钟：读题。 圈出输入、输出、周期、边界和验收要求；退出条件是画出最小数据流和模块清单。

2.            10–25 分钟：建基线。 建立可下载的最小工程；退出条件是 LED、串口和 MSH 正常。

3.            25–70 分钟：通外设。 逐个验证外设；退出条件是每个外设都有独立日志和测试结果。

4.            70–95 分钟：做集成。 接入线程、状态机和显示；退出条件是主流程完整运行。

5.            95–110 分钟：测异常。 检查边界、掉线和错误输入；退出条件是至少完成一种故障恢复。

6.            110–120 分钟：交付。 清理警告、录屏并最终验收；退出条件是系统重启后一次通过演示。
